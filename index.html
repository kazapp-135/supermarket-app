function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheetApp = SpreadsheetApp.getActiveSpreadsheet(); // ========================================== // ①【スーパー専用アプリ】から送信された場合 // ========================================== if (data.type === "super") { var amount = Number(data.value); var now = new Date(); // 現在の時間を自動取得 // 💡シートがなければ自動で作成する安全設計に変更 var superSheet = sheetApp.getSheetByName("スーパー家計簿"); if (!superSheet) { superSheet = sheetApp.insertSheet("スーパー家計簿"); } // 1行目の見出し（A1: 年月日, B1: 円）が空ならセット if (superSheet.getRange(1, 1).getValue() === "") { superSheet.getRange(1, 1).setValue("年月日"); superSheet.getRange(1, 2).setValue("円"); } // 2行目以降に自動取得した日時と金額を追記 superSheet.appendRow([now, amount]); // --- 「金額計算」シート（29行目）への自動集計 --- var kakeiboSheet = sheetApp.getSheetByName("金額計算"); if (kakeiboSheet) { var year = now.getFullYear(); var month = ("0" + (now.getMonth() + 1)).slice(-2); var targetMonth = year + "/" + month; // 例: "2026/05" // 29行目A列に見出しをセット if (kakeiboSheet.getRange(29, 1).getValue() === "") { kakeiboSheet.getRange(29, 1).setValue("食費（スーパー）"); } // 1行目から「現在の年月（例: 2026/05）」の列を探す var lastColumn = kakeiboSheet.getLastColumn(); var targetColumnIndex = -1; for (var col = 2; col <= lastColumn; col++) { var cellValue = kakeiboSheet.getRange(1, col).getDisplayValue(); if (cellValue.indexOf(targetMonth) !== -1 || targetMonth.indexOf(cellValue) !== -1) { targetColumnIndex = col; break; } } // 今月の列が見つかったら、スーパー家計簿の「今月の合計」を計算して29行目を上書き if (targetColumnIndex !== -1) { var superData = superSheet.getDataRange().getValues(); var monthlyTotal = 0; for (var i = 1; i < superData.length; i++) { var dateVal = new Date(superData[i][0]); if (!isNaN(dateVal.getTime())) { var sYear = dateVal.getFullYear(); var sMonth = ("0" + (dateVal.getMonth() + 1)).slice(-2); if (sYear + "/" + sMonth === targetMonth) { monthlyTotal += Number(superData[i][1] || 0); } } } kakeiboSheet.getRange(29, targetColumnIndex).setValue(monthlyTotal); } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // ========================================== // ②【通常の家計簿アプリ】から送信された場合 // ========================================== var targetMonth = data.month; var targetRowName = data.rowName; var value = Number(data.value); var sheet = sheetApp.getSheetByName("金額計算"); var lastColumn = sheet.getLastColumn(); var targetColumnIndex = -1; for (var col = 2; col <= lastColumn; col++) { if (sheet.getRange(1, col).getDisplayValue() === targetMonth) { targetColumnIndex = col; break; } } if (targetColumnIndex === -1) { targetColumnIndex = lastColumn + 1; sheet.getRange(1, targetColumnIndex).setValue(targetMonth); } var lastRow = sheet.getLastRow(); var targetRowIndex = -1; for (var row = 2; row <= lastRow; row++) { if (sheet.getRange(row, 1).getDisplayValue() === targetRowName) { targetRowIndex = row; break; } } if (targetRowIndex !== -1) { sheet.getRange(targetRowIndex, targetColumnIndex).setValue(value); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "項目が見つかりません" })).setMimeType(ContentService.MimeType.JSON); } } catch(error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
